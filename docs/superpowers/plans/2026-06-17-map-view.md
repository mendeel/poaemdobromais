**Goal:** Add a Topbar-toggled map view that plots the current filtered coupons as clustered logo pins on a Porto Alegre map, with a styled preview popup that opens the existing CupomModal.

- [ ] **Step 3: Write minimal implementation**

```js
// src/lib/markers.js
const valid = (v) => typeof v === 'number' && Number.isFinite(v);

// Flatten coupons into one marker per address (endereco) with usable coords.
// Falls back to a coupon's top-level lat/lng when no address has coords.
export function buildMarkers(cupoms) {
  const out = [];
  for (const cupom of cupoms) {
    const pts = (cupom.endereco || [])
      .map((e) => ({ lat: e.latitude, lng: e.longitude }))
      .filter((p) => valid(p.lat) && valid(p.lng));

    const used = pts.length
      ? pts
      : valid(cupom.lat) && valid(cupom.lng)
        ? [{ lat: cupom.lat, lng: cupom.lng }]
        : [];

    used.forEach((p, i) => {
      out.push({ cupom, lat: p.lat, lng: p.lng, key: `${cupom.id}-${i}` });
    });
  }
  return out;
}
```

// src/components/mapIcon.js
import L from 'leaflet';

// Inline styles so we don't depend on Tailwind reaching into Leaflet's DOM.
const RING = '#dc2626';        // --color-primary
const SURFACE = '#ffffff';     // --color-surface

function pinHtml(cupom) {
  const letter = (cupom.nome || '?').trim().charAt(0).toUpperCase();
  const img = cupom.logo
    ? `<img src="${cupom.logo}" alt="" loading="lazy"
         style="width:100%;height:100%;object-fit:cover;border-radius:50%"
         onerror="this.replaceWith(Object.assign(document.createElement('div'),{
           textContent:'${letter}',
           style:'width:100%;height:100%;display:grid;place-items:center;border-radius:50%;background:${RING};color:#fff;font-weight:700;font-family:Karla,sans-serif'
         }))" />`
    : `<div style="width:100%;height:100%;display:grid;place-items:center;border-radius:50%;background:${RING};color:#fff;font-weight:700">${letter}</div>`;

  return `
    <div style="position:relative;width:42px;height:42px">
      <div style="width:42px;height:42px;border-radius:50%;background:${SURFACE};
        border:3px solid ${RING};box-shadow:0 2px 6px rgba(59,26,26,.35);
        overflow:hidden">${img}</div>
      <div style="position:absolute;left:50%;bottom:-6px;transform:translateX(-50%);
        width:0;height:0;border-left:6px solid transparent;border-right:6px solid transparent;
        border-top:8px solid ${RING}"></div>
    </div>`;
}

export function logoDivIcon(cupom) {
  return L.divIcon({
    html: pinHtml(cupom),
    className: '',          // suppress Leaflet's default white box
    iconSize: [42, 50],
    iconAnchor: [21, 50],   // tip of the pointer
    popupAnchor: [0, -50],
  });
}

export function clusterIcon(cluster) {
  const count = cluster.getChildCount();
  return L.divIcon({
    html: `<div style="width:40px;height:40px;border-radius:50%;
      background:${RING};color:#fff;display:grid;place-items:center;
      font-weight:700;font-family:Karla,sans-serif;border:3px solid ${SURFACE};
      box-shadow:0 2px 6px rgba(59,26,26,.35)">${count}</div>`,
    className: '',
    iconSize: [40, 40],
  });
}
```

// src/components/MapView.jsx
import { useMemo } from 'react';
import { MapContainer, TileLayer, Marker, Popup, useMap } from 'react-leaflet';
import MarkerClusterGroup from 'react-leaflet-cluster';
import 'leaflet/dist/leaflet.css';
import { buildMarkers } from '../lib/markers.js';
import { logoDivIcon, clusterIcon } from './mapIcon.js';

const POA_CENTER = [-30.0346, -51.2177];

// Pan/zoom to fit the current markers whenever they change.
function FitBounds({ markers }) {
  const map = useMap();
  useMemo(() => {
    if (!markers.length) return;
    const bounds = markers.map((m) => [m.lat, m.lng]);
    map.fitBounds(bounds, { padding: [48, 48], maxZoom: 15 });
  }, [markers, map]);
  return null;
}

export default function MapView({ cupoms, onOpen }) {
  const markers = useMemo(() => buildMarkers(cupoms), [cupoms]);

  return (
    <div className="min-h-0 flex-1">
      <MapContainer
        center={POA_CENTER}
        zoom={12}
        scrollWheelZoom
        style={{ height: '100%', width: '100%' }}
      >
        <TileLayer
          attribution='&copy; <a href="https://www.openstreetmap.org/copyright">OSM</a> &copy; <a href="https://carto.com/attributions">CARTO</a>'
          url="https://{s}.basemaps.cartocdn.com/rastertiles/voyager/{z}/{x}/{y}{r}.png"
        />
        <FitBounds markers={markers} />
        <MarkerClusterGroup iconCreateFunction={clusterIcon} showCoverageOnHover={false}>
          {markers.map((m) => (
            <Marker key={m.key} position={[m.lat, m.lng]} icon={logoDivIcon(m.cupom)}>
              <Popup>
                <div style={{ width: 200, fontFamily: 'Karla, sans-serif' }}>
                  <div style={{ display: 'flex', gap: 8, alignItems: 'center' }}>
                    {m.cupom.logo && (
                      <img
                        src={m.cupom.logo}
                        alt=""
                        style={{ width: 36, height: 36, borderRadius: 8, objectFit: 'cover', flexShrink: 0 }}
                        onError={(e) => { e.currentTarget.style.display = 'none'; }}
                      />
                    )}
                    <strong style={{ color: '#3b1a1a', fontSize: 14 }}>{m.cupom.nome}</strong>
                  </div>
                  <div style={{ color: '#dc2626', fontWeight: 700, fontSize: 13, marginTop: 6 }}>
                    {m.cupom.beneficio}
                  </div>
                  {m.cupom.endereco?.[0]?.bairro && (
                    <div style={{ color: '#8a6f6f', fontSize: 12, marginTop: 2 }}>
                      {m.cupom.endereco[0].bairro}
                    </div>
                  )}
                  <button
                    type="button"
                    onClick={() => onOpen(m.cupom)}
                    style={{
                      marginTop: 8, width: '100%', padding: '7px 0', border: 'none',
                      borderRadius: 9, background: '#dc2626', color: '#fff',
                      fontWeight: 700, cursor: 'pointer', fontFamily: 'Karla, sans-serif',
                    }}
                  >
                    Ver cupom
                  </button>
                </div>
              </Popup>
            </Marker>
          ))}
        </MarkerClusterGroup>
      </MapContainer>
    </div>
  );
}
```
